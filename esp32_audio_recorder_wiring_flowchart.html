Power Bus Note (1× 3V3 & Multiple GND)

Your ESP32 DevKit provides one 3V3 pin and multiple GND pins (all tied together internally). Route 3V3 to a breadboard power rail to feed both the INMP441 VDD and MicroSD VCC (use VIN if your SD board has a 3.3V regulator), and tie all GND pins together with the Push Button (INPUT_PULLUP on D4 to GND).

esp32_button_wav_recorder.ino • Arduino IDE / PlatformIO 
#include <Arduino.h>
#include <SPI.h>
#include <SD.h>
#include <driver/i2s.h>

// ==========================================
// 1. PIN DEFINITIONS (Matches Wiring Chart)
// ==========================================
// Push Button (Toggle Record / Stop) -> Silkscreen D4 (GPIO4) to GND
#define BUTTON_PIN      4   // D4

// MicroSD Card Module (VSPI) -> Silkscreen D5, D23, D19, D18
#define SD_CS_PIN       5   // D5
#define SD_MOSI_PIN     23  // D23
#define SD_MISO_PIN     19  // D19
#define SD_SCK_PIN      18  // D18

// INMP441 I2S Microphone -> Silkscreen D25, D26, D33
#define I2S_WS_PIN      25  // D25
#define I2S_SCK_PIN     26  // D26
#define I2S_SD_PIN      33  // D33
#define I2S_PORT        I2S_NUM_0

// Audio Sampling Configuration
#define SAMPLE_RATE     16000
#define BITS_PER_SAMPLE 16
#define CHANNELS        1

File audioFile;
bool isRecording = false;
uint32_t dataBytesWritten = 0;
int fileIndex = 1;

// Debounce variables
int lastButtonState = HIGH;
int stableButtonState = HIGH;
unsigned long lastDebounceTime = 0;
const unsigned long debounceDelay = 50;

void writeWavHeader(File &file, uint32_t sampleRate, uint16_t bitsPerSample, uint16_t channels, uint32_t dataSize) {
  uint32_t byteRate = sampleRate * channels * (bitsPerSample / 8);
  uint16_t blockAlign = channels * (bitsPerSample / 8);
  uint32_t chunkSize = 36 + dataSize;

  uint8_t header[44] = {
    'R','I','F','F',
    (uint8_t)(chunkSize & 0xFF), (uint8_t)((chunkSize >> 8) & 0xFF), (uint8_t)((chunkSize >> 16) & 0xFF), (uint8_t)((chunkSize >> 24) & 0xFF),
    'W','A','V','E',
    'f','m','t',' ',
    16, 0, 0, 0,
    1, 0,
    (uint8_t)(channels & 0xFF), (uint8_t)((channels >> 8) & 0xFF),
    (uint8_t)(sampleRate & 0xFF), (uint8_t)((sampleRate >> 8) & 0xFF), (uint8_t)((sampleRate >> 16) & 0xFF), (uint8_t)((sampleRate >> 24) & 0xFF),
    (uint8_t)(byteRate & 0xFF), (uint8_t)((byteRate >> 8) & 0xFF), (uint8_t)((byteRate >> 16) & 0xFF), (uint8_t)((byteRate >> 24) & 0xFF),
    (uint8_t)(blockAlign & 0xFF), (uint8_t)((blockAlign >> 8) & 0xFF),
    (uint8_t)(bitsPerSample & 0xFF), (uint8_t)((bitsPerSample >> 8) & 0xFF),
    'd','a','t','a',
    (uint8_t)(dataSize & 0xFF), (uint8_t)((dataSize >> 8) & 0xFF), (uint8_t)((dataSize >> 16) & 0xFF), (uint8_t)((dataSize >> 24) & 0xFF)
  };

  file.seek(0);
  file.write(header, 44);
}

void setupI2S() {
  i2s_config_t i2s_config = {
    .mode = (i2s_mode_t)(I2S_MODE_MASTER | I2S_MODE_RX),
    .sample_rate = SAMPLE_RATE,
    .bits_per_sample = I2S_BITS_PER_SAMPLE_32BIT,
    .channel_format = I2S_CHANNEL_FMT_ONLY_LEFT,
    .communication_format = I2S_COMM_FORMAT_STAND_I2S,
    .intr_alloc_flags = ESP_INTR_FLAG_LEVEL1,
    .dma_buf_count = 8,
    .dma_buf_len = 256,
    .use_apll = false,
    .tx_desc_auto_clear = false,
    .fixed_mclk = 0
  };

  i2s_pin_config_t pin_config = {
    .bck_io_num = I2S_SCK_PIN,
    .ws_io_num = I2S_WS_PIN,
    .data_out_num = I2S_PIN_NO_CHANGE,
    .data_in_num = I2S_SD_PIN
  };

  i2s_driver_install(I2S_PORT, &i2s_config, 0, NULL);
  i2s_set_pin(I2S_PORT, &pin_config);
}

void startRecording() {
  char filename[24];
  while (true) {
    snprintf(filename, sizeof(filename), "/rec_%03d.wav", fileIndex++);
    if (!SD.exists(filename)) break;
  }

  audioFile = SD.open(filename, FILE_WRITE);
  if (!audioFile) {
    Serial.println("ERROR: Could not open file on SD card.");
    return;
  }

  uint8_t emptyHeader[44] = {0};
  audioFile.write(emptyHeader, 44);
  dataBytesWritten = 0;
  isRecording = true;
  Serial.print("Recording started: ");
  Serial.println(filename);
}

void stopRecording() {
  if (!isRecording) return;
  isRecording = false;
  writeWavHeader(audioFile, SAMPLE_RATE, BITS_PER_SAMPLE, CHANNELS, dataBytesWritten);
  audioFile.flush();
  audioFile.close();
  Serial.print("Recording saved. Total bytes: ");
  Serial.println(dataBytesWritten);
}

void setup() {
  Serial.begin(115200);
  pinMode(BUTTON_PIN, INPUT_PULLUP);

  SPI.begin(SD_SCK_PIN, SD_MISO_PIN, SD_MOSI_PIN, SD_CS_PIN);
  if (!SD.begin(SD_CS_PIN)) {
    Serial.println("ERROR: SD Card Mount Failed! Check wiring.");
  } else {
    Serial.println("SD Card initialized.");
  }

  setupI2S();
  Serial.println("System Ready. Press button on D4 (GPIO4) to record.");
}

void loop() {
  int reading = digitalRead(BUTTON_PIN);
  if (reading != lastButtonState) {
    lastDebounceTime = millis();
  }

  if ((millis() - lastDebounceTime) > debounceDelay) {
    if (reading != stableButtonState) {
      stableButtonState = reading;
      if (stableButtonState == LOW) {
        if (!isRecording) startRecording();
        else stopRecording();
      }
    }
  }
  lastButtonState = reading;

  if (isRecording && audioFile) {
    int32_t rawSamples[256];
    size_t bytesRead = 0;
    i2s_read(I2S_PORT, &rawSamples, sizeof(rawSamples), &bytesRead, portMAX_DELAY);

    int sampleCount = bytesRead / sizeof(int32_t);
    int16_t pcm16[256];
    for (int i = 0; i < sampleCount; i++) {
      pcm16[i] = (int16_t)(rawSamples[i] >> 16);
    }

    size_t outBytes = sampleCount * sizeof(int16_t);
    audioFile.write((uint8_t*)pcm16, outBytes);
    dataBytesWritten += outBytes;
  }
}
